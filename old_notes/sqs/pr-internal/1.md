package com.example.sqs.service;

import com.example.sqs.entity.MessageEntity;
import com.example.sqs.util.SqsUtil;
import software.amazon.awssdk.services.sqs.SqsClient;
import software.amazon.awssdk.services.sqs.SqsAsyncClient;
import software.amazon.awssdk.services.sqs.model.*;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import java.util.List;
import java.util.concurrent.CompletableFuture;
import java.util.stream.Collectors;

/**
 * AWS SQS Producer Service - Handles sending messages to SQS queues
 * Uses AWS SDK 2.0 with both synchronous and asynchronous operations
 */
@Service
public class SqsProducerService {
    
    private static final Logger logger = LoggerFactory.getLogger(SqsProducerService.class);
    
    @Autowired
    private SqsClient sqsClient;
    
    @Autowired
    private SqsAsyncClient sqsAsyncClient;
    
    @Autowired
    private SqsUtil sqsUtil;

    /**
     * Send a single message to SQS queue synchronously
     * @param queueName - Name of the SQS queue
     * @param messageEntity - Message entity containing message data
     * @return SendMessageResponse
     */
    public SendMessageResponse sendMessage(String queueName, MessageEntity messageEntity) {
        try {
            // Get queue URL using utility method
            String queueUrl = sqsUtil.getQueueUrl(queueName);
            
            // Build send message request
            SendMessageRequest request = SendMessageRequest.builder()
                    .queueUrl(queueUrl)
                    .messageBody(messageEntity.getBody())
                    .messageAttributes(sqsUtil.convertToMessageAttributes(messageEntity.getAttributes()))
                    .delaySeconds(messageEntity.getDelaySeconds())
                    .build();

            // Send message synchronously
            SendMessageResponse response = sqsClient.sendMessage(request);
            
            logger.info("Message sent successfully. MessageId: {}", response.messageId());
            return response;
            
        } catch (Exception e) {
            logger.error("Error sending message to queue: {}", queueName, e);
            throw new RuntimeException("Failed to send message", e);
        }
    }

    /**
     * Send a single message to SQS queue asynchronously
     * @param queueName - Name of the SQS queue
     * @param messageEntity - Message entity containing message data
     * @return CompletableFuture<SendMessageResponse>
     */
    public CompletableFuture<SendMessageResponse> sendMessageAsync(String queueName, MessageEntity messageEntity) {
        try {
            // Get queue URL using utility method
            String queueUrl = sqsUtil.getQueueUrl(queueName);
            
            // Build send message request
            SendMessageRequest request = SendMessageRequest.builder()
                    .queueUrl(queueUrl)
                    .messageBody(messageEntity.getBody())
                    .messageAttributes(sqsUtil.convertToMessageAttributes(messageEntity.getAttributes()))
                    .delaySeconds(messageEntity.getDelaySeconds())
                    .build();

            // Send message asynchronously and handle response
            return sqsAsyncClient.sendMessage(request)
                    .whenComplete((response, throwable) -> {
                        if (throwable != null) {
                            logger.error("Error sending message async to queue: {}", queueName, throwable);
                        } else {
                            logger.info("Message sent async successfully. MessageId: {}", response.messageId());
                        }
                    });
            
        } catch (Exception e) {
            logger.error("Error in async send message setup for queue: {}", queueName, e);
            CompletableFuture<SendMessageResponse> failedFuture = new CompletableFuture<>();
            failedFuture.completeExceptionally(e);
            return failedFuture;
        }
    }

    /**
     * Send multiple messages to SQS queue in batch synchronously
     * @param queueName - Name of the SQS queue
     * @param messageEntities - List of message entities
     * @return SendMessageBatchResponse
     */
    public SendMessageBatchResponse sendMessageBatch(String queueName, List<MessageEntity> messageEntities) {
        try {
            // Get queue URL using utility method
            String queueUrl = sqsUtil.getQueueUrl(queueName);
            
            // Convert message entities to batch request entries
            List<SendMessageBatchRequestEntry> entries = messageEntities.stream()
                    .map(entity -> SendMessageBatchRequestEntry.builder()
                            .id(entity.getId()) // Unique ID for each message in batch
                            .messageBody(entity.getBody())
                            .messageAttributes(sqsUtil.convertToMessageAttributes(entity.getAttributes()))
                            .delaySeconds(entity.getDelaySeconds())
                            .build())
                    .collect(Collectors.toList());

            // Build batch send request
            SendMessageBatchRequest request = SendMessageBatchRequest.builder()
                    .queueUrl(queueUrl)
                    .entries(entries)
                    .build();

            // Send batch synchronously
            SendMessageBatchResponse response = sqsClient.sendMessageBatch(request);
            
            logger.info("Batch sent successfully. Successful: {}, Failed: {}", 
                       response.successful().size(), response.failed().size());
            
            // Log any failed messages
            response.failed().forEach(failed -> 
                logger.warn("Failed to send message with ID: {} - Reason: {}", 
                           failed.id(), failed.message()));
            
            return response;
            
        } catch (Exception e) {
            logger.error("Error sending message batch to queue: {}", queueName, e);
            throw new RuntimeException("Failed to send message batch", e);
        }
    }

    /**
     * Send multiple messages to SQS queue in batch asynchronously
     * @param queueName - Name of the SQS queue
     * @param messageEntities - List of message entities
     * @return CompletableFuture<SendMessageBatchResponse>
     */
    public CompletableFuture<SendMessageBatchResponse> sendMessageBatchAsync(String queueName, List<MessageEntity> messageEntities) {
        try {
            // Get queue URL using utility method
            String queueUrl = sqsUtil.getQueueUrl(queueName);
            
            // Convert message entities to batch request entries
            List<SendMessageBatchRequestEntry> entries = messageEntities.stream()
                    .map(entity -> SendMessageBatchRequestEntry.builder()
                            .id(entity.getId())
                            .messageBody(entity.getBody())
                            .messageAttributes(sqsUtil.convertToMessageAttributes(entity.getAttributes()))
                            .delaySeconds(entity.getDelaySeconds())
                            .build())
                    .collect(Collectors.toList());

            // Build batch send request
            SendMessageBatchRequest request = SendMessageBatchRequest.builder()
                    .queueUrl(queueUrl)
                    .entries(entries)
                    .build();

            // Send batch asynchronously and handle response
            return sqsAsyncClient.sendMessageBatch(request)
                    .whenComplete((response, throwable) -> {
                        if (throwable != null) {
                            logger.error("Error sending message batch async to queue: {}", queueName, throwable);
                        } else {
                            logger.info("Batch sent async successfully. Successful: {}, Failed: {}", 
                                       response.successful().size(), response.failed().size());
                            
                            // Log any failed messages
                            response.failed().forEach(failed -> 
                                logger.warn("Failed to send message with ID: {} - Reason: {}", 
                                           failed.id(), failed.message()));
                        }
                    });
            
        } catch (Exception e) {
            logger.error("Error in async send batch setup for queue: {}", queueName, e);
            CompletableFuture<SendMessageBatchResponse> failedFuture = new CompletableFuture<>();
            failedFuture.completeExceptionally(e);
            return failedFuture;
        }
    }

    /**
     * Get queue attributes synchronously
     * @param queueName - Name of the SQS queue
     * @param attributeNames - List of attribute names to retrieve
     * @return GetQueueAttributesResponse
     */
    public GetQueueAttributesResponse getQueueAttributes(String queueName, List<QueueAttributeName> attributeNames) {
        try {
            // Get queue URL using utility method
            String queueUrl = sqsUtil.getQueueUrl(queueName);
            
            // Build get queue attributes request
            GetQueueAttributesRequest request = GetQueueAttributesRequest.builder()
                    .queueUrl(queueUrl)
                    .attributeNames(attributeNames)
                    .build();

            // Get attributes synchronously
            GetQueueAttributesResponse response = sqsClient.getQueueAttributes(request);
            
            logger.info("Retrieved queue attributes for queue: {}", queueName);
            return response;
            
        } catch (Exception e) {
            logger.error("Error getting queue attributes for queue: {}", queueName, e);
            throw new RuntimeException("Failed to get queue attributes", e);
        }
    }

    /**
     * Get queue attributes asynchronously
     * @param queueName - Name of the SQS queue
     * @param attributeNames - List of attribute names to retrieve
     * @return CompletableFuture<GetQueueAttributesResponse>
     */
    public CompletableFuture<GetQueueAttributesResponse> getQueueAttributesAsync(String queueName, List<QueueAttributeName> attributeNames) {
        try {
            // Get queue URL using utility method
            String queueUrl = sqsUtil.getQueueUrl(queueName);
            
            // Build get queue attributes request
            GetQueueAttributesRequest request = GetQueueAttributesRequest.builder()
                    .queueUrl(queueUrl)
                    .attributeNames(attributeNames)
                    .build();

            // Get attributes asynchronously and handle response
            return sqsAsyncClient.getQueueAttributes(request)
                    .whenComplete((response, throwable) -> {
                        if (throwable != null) {
                            logger.error("Error getting queue attributes async for queue: {}", queueName, throwable);
                        } else {
                            logger.info("Retrieved queue attributes async for queue: {}", queueName);
                        }
                    });
            
        } catch (Exception e) {
            logger.error("Error in async get queue attributes setup for queue: {}", queueName, e);
            CompletableFuture<GetQueueAttributesResponse> failedFuture = new CompletableFuture<>();
            failedFuture.completeExceptionally(e);
            return failedFuture;
        }
    }
}
